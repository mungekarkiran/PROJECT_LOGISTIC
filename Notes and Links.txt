def hash_password(password: str) -> str:
    return bcrypt.hashpw(password.encode(), bcrypt.gensalt()).decode()

def verify_password(password: str, hashed: str) -> bool:
    try:
        return bcrypt.checkpw(password.encode(), hashed.encode())
    except Exception:
        return False

def create_user(username, email, password, role):
    """Returns (success, message)."""
    conn = get_connection()
    cur = conn.cursor()
    try:
        cur.execute(
            "INSERT INTO users (username, email, password_hash, role, created_at) VALUES (?,?,?,?,?)",
            (username, email, hash_password(password), role, datetime.now().isoformat())
        )
        conn.commit()
        return True, "Registration successful! Please log in."
    except sqlite3.IntegrityError as e:
        if "username" in str(e):
            return False, "Username already exists."
        if "email" in str(e):
            return False, "Email already registered."
        return False, "Registration failed."
    finally:
        conn.close()

def authenticate(username, password):
    """Returns (user_dict, None) on success, (None, error) on failure."""
    conn = get_connection()
    cur = conn.cursor()
    cur.execute("SELECT * FROM users WHERE username = ?", (username,))
    row = cur.fetchone()
    conn.close()

    if row is None:
        return None, "User not found."
    if not verify_password(password, row["password_hash"]):
        return None, "Incorrect password."

    return {
        "id": row["id"],
        "username": row["username"],
        "email": row["email"],
        "role": row["role"],
    }, None

import sqlite3
import bcrypt
from datetime import datetime

DB_PATH = "logistics.db"

def get_connection():
    conn = sqlite3.connect(DB_PATH, check_same_thread=False)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA foreign_keys = ON;")   # enforce FK relationships
    return conn

def init_db():
    """Create all tables (idempotent) and seed a default admin."""
    conn = get_connection()
    cur = conn.cursor()

    # ---------- users ----------
    cur.execute("""
        CREATE TABLE IF NOT EXISTS users (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            username TEXT UNIQUE NOT NULL,
            email TEXT UNIQUE NOT NULL,
            password_hash TEXT NOT NULL,
            role TEXT NOT NULL CHECK(role IN ('Admin','Operator','Planner')),
            created_at TEXT NOT NULL
        )
    """)

    # ---------- warehouses ----------
    cur.execute("""
        CREATE TABLE IF NOT EXISTS warehouses (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL,
            city TEXT NOT NULL,
            state TEXT NOT NULL,
            lat REAL NOT NULL,
            lon REAL NOT NULL
        )
    """)

    # ---------- carriers ----------
    cur.execute("""
        CREATE TABLE IF NOT EXISTS carriers (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL,
            reliability_score REAL NOT NULL,   -- 0.0 - 1.0
            base_rate_per_km REAL NOT NULL
        )
    """)

    # ---------- vehicles ----------
    cur.execute("""
        CREATE TABLE IF NOT EXISTS vehicles (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            plate TEXT UNIQUE NOT NULL,
            carrier_id INTEGER NOT NULL,
            make TEXT,
            model TEXT,
            year INTEGER,
            mileage_km INTEGER NOT NULL,
            last_service_date TEXT NOT NULL,
            FOREIGN KEY (carrier_id) REFERENCES carriers(id)
        )
    """)

    # ---------- shipments ----------
    cur.execute("""
        CREATE TABLE IF NOT EXISTS shipments (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            tracking_no TEXT UNIQUE NOT NULL,
            origin_wh_id INTEGER NOT NULL,
            dest_wh_id INTEGER NOT NULL,
            carrier_id INTEGER NOT NULL,
            vehicle_id INTEGER,
            weight_kg REAL NOT NULL,
            priority TEXT NOT NULL CHECK(priority IN ('Standard','Express','Priority')),
            created_at TEXT NOT NULL,
            promised_date TEXT NOT NULL,
            actual_delivery_date TEXT,           -- NULL until delivered
            status TEXT NOT NULL CHECK(status IN ('Created','Picked Up','In Transit','Out for Delivery','Delivered','Delayed','Cancelled')),
            created_by_user_id INTEGER,
            distance_km REAL,                    -- computed at insert
            eta_hours REAL,                      -- from model (or rule-based)
            FOREIGN KEY (origin_wh_id) REFERENCES warehouses(id),
            FOREIGN KEY (dest_wh_id)   REFERENCES warehouses(id),
            FOREIGN KEY (carrier_id)   REFERENCES carriers(id),
            FOREIGN KEY (vehicle_id)   REFERENCES vehicles(id),
            FOREIGN KEY (created_by_user_id) REFERENCES users(id)
        )
    """)

    # ---------- model_metrics (for Admin page later) ----------
    cur.execute("""
        CREATE TABLE IF NOT EXISTS model_metrics (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            model_name TEXT NOT NULL,
            version TEXT NOT NULL,
            metric_name TEXT NOT NULL,
            metric_value REAL NOT NULL,
            trained_at TEXT NOT NULL,
            notes TEXT
        )
    """)

    # ---------- seed default admin ----------
    cur.execute("SELECT COUNT(*) as c FROM users WHERE role='Admin'")
    if cur.fetchone()["c"] == 0:
        hashed = bcrypt.hashpw("admin123".encode(), bcrypt.gensalt()).decode()
        cur.execute(
            "INSERT INTO users (username, email, password_hash, role, created_at) VALUES (?,?,?,?,?)",
            ("admin", "admin@logistics.local", hashed, "Admin", datetime.now().isoformat())
        )

    conn.commit()
    conn.close()

import streamlit as st
import pandas as pd
from database import get_connection

BLUE  = "#1565C0"
GREEN = "#2E7D32"
DARK  = "#0D47A1"


def render():
    st.markdown(f"<h1 style='color:{DARK};'>🛡️ Admin Dashboard</h1>",
                unsafe_allow_html=True)
    st.caption("Full system access — manage users, view all KPIs and model health.")

    conn = get_connection()
    users = pd.read_sql_query(
        "SELECT id, username, email, role, created_at FROM users ORDER BY id",
        conn,
    )
    conn.close()

    # KPI row
    c1, c2, c3, c4 = st.columns(4)
    c1.metric("Total Users", len(users))
    c2.metric("Admins",    int((users["role"] == "Admin").sum()))
    c3.metric("Operators", int((users["role"] == "Operator").sum()))
    c4.metric("Planners",  int((users["role"] == "Planner").sum()))

    st.divider()
    st.subheader("👥 Registered Users")
    st.dataframe(users, use_container_width=True)

    # ---------- Model Health ----------
    st.divider()
    st.subheader("🤖 Model Health")
    try:
        conn = get_connection()
        mm = pd.read_sql_query(
            "SELECT model_name, version, metric_name, metric_value, trained_at "
            "FROM model_metrics ORDER BY trained_at DESC LIMIT 12",
            conn,
        )
        conn.close()
    except Exception as e:
        mm = pd.DataFrame()
        st.info(f"Model metrics not available yet ({e}).")

    if mm.empty:
        st.info(
            "No models trained yet. Run `notebooks/eta_model_training.ipynb` "
            "to train the ETA model and populate this table."
        )
    else:
        mm["metric_value"] = mm["metric_value"].round(3)
        st.dataframe(mm, use_container_width=True)

        # small visual: latest metrics per model
        latest = (mm.sort_values("trained_at")
                    .groupby(["model_name", "metric_name"], as_index=False)
                    .last())
        if not latest.empty:
            st.markdown("**Latest metrics per model**")
            pivot = latest.pivot(index="model_name",
                                 columns="metric_name",
                                 values="metric_value")
            st.dataframe(pivot, use_container_width=True)

    # ---------- Coming Soon ----------
    st.divider()
    st.subheader("🚀 Coming Soon in Admin")
    st.info(
        "• System-wide KPI overview (OTD, OTIF, cost per shipment) — reads from shipments\n\n"
        "• Audit trail and role-permission management\n\n"
        "• Drill-down into any Operator / Planner action\n\n"
        "• Trigger model retraining directly from this dashboard"
    )
